\begin{columns}
  % ---- Column 1: Context & Core Questions ----
  \column{0.333}
  
  \block{1. Introduction \& Problem Statement}{
    \textbf{Operational Context:}
    \begin{itemize}\setlength{\itemsep}{4pt}
      \item Marine networks (EMODnet, SeaDataNet) require controlled vocabularies (e.g. BODC P02), yet non-English coverage remains fragmented.
      \item Spreadsheets impede cross-border discovery and FAIR reuse.
    \end{itemize}
    
    \vspace{0.25cm}
    \textbf{Research Question (RQ):}
    \begin{itemize}\setlength{\itemsep}{3pt}
      \item \textit{How can marine institutes independently manage multilingual vocabularies while ensuring machine-actionable FAIR interoperability?}
    \end{itemize}
    
    \vspace{0.25cm}
    \textbf{Hypothesis \& Prediction:}
    \begin{itemize}\setlength{\itemsep}{3pt}
      \item \textit{Decoupled concept modeling combined with push-based LDES event streams lowers synchronization latency without sacrificing curation quality.}
    \end{itemize}
  }
  
  \block{2. FAIR Architectural Pillars}{
    \begin{itemize}\setlength{\itemsep}{4pt}
      \item \textbf{Findable:} Persistent URIs with multilingual SKOS labels.
      \item \textbf{Accessible:} Dual exports in open formats (CSV, JSON-LD, RDF).
      \item \textbf{Interoperable:} LDES event feeds enable live machine sync.
      \item \textbf{Reusable:} ORCID-anchored contributor provenance.
    \end{itemize}
  }

  % ---- Column 2: Methods & Pipeline ----
  \column{0.334}
  
  \block{3. Methodology \& Technical Stack}{
    \begin{itemize}\setlength{\itemsep}{4pt}
      \item \textbf{Frontend UI (React):} Side-by-side comparison and review gates.
      \item \textbf{API Services (Node.js):} Enforces consistency constraints.
      \item \textbf{Persistence (PostgreSQL):} Isolates URIs from labels.
      \item \textbf{Identity:} ORCID OAuth2 login ensuring contributor provenance.
    \end{itemize}
  }
  
  \block{4. Curatorial Workflow Pipeline}{

        };

  % ---- Column 3: Results & Synthesis ----
  \column{0.333}
  
  \block{5. Results: BODC P02 Beta Evaluation}{
    \begin{itemize}\setlength{\itemsep}{4pt}
      \item \textbf{25\% of BODC P02 Translated:} Enriched parameter discovery concepts into French and Dutch during live beta.
      \item \textbf{Continuous LDES Stream:} Changes published as versioned RDF event feeds without monolithic snapshot downloads.
      \item \textbf{Zero Concept URI Churn:} Persistent URIs remained stable across translation updates.
      \item \textbf{2$\times$ Curation Speed:} AI suggestions halved drafting time under strict human moderation.
    \end{itemize}
  }

    \vspace{0.25cm}
